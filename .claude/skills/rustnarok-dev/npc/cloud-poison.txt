// Cloud Kill's poison (issue #793): whisper npc:RustnarokCloud "poison" to start SC_CLOUD_POISON on the player, "clear"
// to end it. Loaded and unloaded by scenarios/earth-drive-cloud-poison.json.
-	script	RustnarokCloud	-1,{
	end;
OnWhisperGlobal:
	if (@whispervar0$ == "poison") {
		sc_start SC_CLOUD_POISON, 600000, 1;
	} else if (@whispervar0$ == "clear") {
		sc_end SC_CLOUD_POISON;
	}
	end;
}
