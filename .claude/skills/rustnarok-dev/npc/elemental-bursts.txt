// The elementals' bursts (issue #817), on prt_fild08. Whisper npc:RustnarokBursts "fire", "wind" or "earth" and the elementals'
// burst (936 - 938) plays on the player, since Pyrotechnic, Gust and Petrology only come from an elemental. Loaded and unloaded
// by scenarios/elemental-bursts.json.
prt_fild08,1,1,0	script	RustnarokBursts	-1,{
	end;
OnWhisperGlobal:
	if (@whispervar0$ == "fire") {
		specialeffect2 936;
	} else if (@whispervar0$ == "wind") {
		specialeffect2 937;
	} else if (@whispervar0$ == "earth") {
		specialeffect2 938;
	}
	end;
}
