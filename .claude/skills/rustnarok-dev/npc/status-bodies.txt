// Status bodies (issue #452): whisper npc:RustnarokBodies "assumptio", "steel", "berserk" or "song" to start Assumptio,
// Steel Body, Berserk or the Poem of Bragi on the player, "clear" to end them. Loaded and unloaded by
// scenarios/status-bodies.json.
-	script	RustnarokBodies	-1,{
	end;
OnWhisperGlobal:
	if (@whispervar0$ == "assumptio") {
		sc_start SC_ASSUMPTIO, 600000, 5;
	} else if (@whispervar0$ == "steel") {
		sc_start SC_STEELBODY, 600000, 5;
	} else if (@whispervar0$ == "berserk") {
		sc_start SC_BERSERK, 600000, 1;
	} else if (@whispervar0$ == "song") {
		sc_start SC_POEMBRAGI, 600000, 10;
	} else if (@whispervar0$ == "clear") {
		sc_end SC_ASSUMPTIO;
		sc_end SC_STEELBODY;
		sc_end SC_BERSERK;
		sc_end SC_POEMBRAGI;
	}
	end;
}
